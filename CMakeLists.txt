cmake_minimum_required(VERSION 3.20)
project(ImperivmMapExtractor VERSION 0.2.3 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the MSVC x64 tools on Windows")
endif()
set(IBERIVM_SRC "${CMAKE_CURRENT_SOURCE_DIR}/vendor/iberivm/src" CACHE PATH "Bundled generator headers")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(mapx_core INTERFACE)
target_compile_features(mapx_core INTERFACE cxx_std_20)
target_include_directories(mapx_core INTERFACE src "${IBERIVM_SRC}/data" "${IBERIVM_SRC}/contract"
    "${IBERIVM_SRC}/sim" "${IBERIVM_SRC}/sim/randommap")
target_compile_options(mapx_core INTERFACE /W4 /utf-8)
add_executable(imperivm-map-extractor WIN32 src/main.cpp src/app.rc src/app.manifest)
target_link_libraries(imperivm-map-extractor PRIVATE mapx_core user32 gdi32 comdlg32 shell32 ole32 uuid advapi32 comctl32)
target_compile_definitions(imperivm-map-extractor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(imperivm-map-extractor PRIVATE /bigobj $<$<CONFIG:Release>:/GL /Gy>)
target_link_options(imperivm-map-extractor PRIVATE $<$<CONFIG:Release>:/LTCG /OPT:REF /OPT:ICF /DEBUG:NONE>)
add_executable(scrub_binary tools/scrub_binary.cpp)
add_custom_command(TARGET imperivm-map-extractor POST_BUILD
    COMMAND scrub_binary "$<TARGET_FILE:imperivm-map-extractor>"
    COMMENT "Neutralising address texts in the executable")
option(MAPX_BUILD_DEVELOPER_TOOLS "Build the console probe and performance utility" ON)
if(MAPX_BUILD_DEVELOPER_TOOLS)
    add_executable(map-extractor-probe tools/probe.cpp)
    target_link_libraries(map-extractor-probe PRIVATE mapx_core)
    add_executable(map-generation-benchmark tools/generation_benchmark.cpp)
    target_link_libraries(map-generation-benchmark PRIVATE mapx_core)
    target_compile_definitions(map-generation-benchmark PRIVATE OPTIMIZED_GENERATION)
    target_compile_options(map-generation-benchmark PRIVATE /O2)
endif()
include(CTest)
if(BUILD_TESTING)
    foreach(name IN ITEMS input_reader pre_session hpfs_writer minimap)
        add_executable(${name}_test tests/${name}_test.cpp)
        target_link_libraries(${name}_test PRIVATE mapx_core)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    add_executable(release_safety_test tests/release_safety_test.cpp)
    target_link_libraries(release_safety_test PRIVATE mapx_core shell32 ole32 uuid)
    target_compile_definitions(release_safety_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME release_safety COMMAND release_safety_test)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME release_strings COMMAND Python3::Interpreter -I "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_strings.py"
            "$<TARGET_FILE:imperivm-map-extractor>")
    endif()
    set(MAPX_GAME_ROOT "" CACHE PATH "Optional compatible game installation for integration tests")
    set(MAPX_REPLAY_FIXTURE "" CACHE FILEPATH "Optional owner-provided replay for UI tests")
    set(MAPX_SAVE_FIXTURE_A "" CACHE FILEPATH "Optional owner-provided verified save")
    set(MAPX_SAVE_FIXTURE_B "" CACHE FILEPATH "Optional owner-provided mismatch save")
    set(MAPX_FIXTURE_BASE "" CACHE PATH "Optional external fixture root")
    set(MAPX_FIXTURE_MANIFEST_DIR "" CACHE PATH "Optional external frozen fixture manifests")
    if(MAPX_GAME_ROOT)
        foreach(name IN ITEMS gallery gallery_matrix)
            add_executable(${name}_test tests/${name}_test.cpp)
            target_link_libraries(${name}_test PRIVATE mapx_core)
            add_test(NAME ${name} COMMAND ${name}_test "${MAPX_GAME_ROOT}")
            set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
        endforeach()
        add_executable(randommap_pipeline_test tests/randommap_pipeline_test.cpp)
        target_link_libraries(randommap_pipeline_test PRIVATE mapx_core)
        add_test(NAME generation_pipeline COMMAND randommap_pipeline_test "${MAPX_GAME_ROOT}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cases/e05_continent.json"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cases/e10_mountain_winter.json"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cases/e14_continent_flags.json"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cases/e17_island.json")
        set_tests_properties(generation_pipeline PROPERTIES SKIP_RETURN_CODE 77)
        if(MAPX_SAVE_FIXTURE_A AND MAPX_SAVE_FIXTURE_B)
            add_executable(generation_optimization_test tests/generation_optimization_test.cpp)
            target_link_libraries(generation_optimization_test PRIVATE mapx_core)
            add_test(NAME generation_optimization COMMAND generation_optimization_test "${MAPX_GAME_ROOT}"
                "${MAPX_SAVE_FIXTURE_A}" "${MAPX_SAVE_FIXTURE_B}")
            set_tests_properties(generation_optimization PROPERTIES SKIP_RETURN_CODE 77)
        endif()
        if(Python3_FOUND)
            add_test(NAME error_ui COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_error_ui.py"
                --exe "$<TARGET_FILE:imperivm-map-extractor>" --game "${MAPX_GAME_ROOT}"
                --out "${CMAKE_CURRENT_BINARY_DIR}/error-ui")
            set_tests_properties(error_ui PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120 RUN_SERIAL TRUE)
            if(MAPX_REPLAY_FIXTURE)
                add_test(NAME gallery_ui COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_gallery_ui.py"
                    --exe "$<TARGET_FILE:imperivm-map-extractor>" --core-test "$<TARGET_FILE:gallery_test>"
                    --game "${MAPX_GAME_ROOT}" --fixture "${MAPX_REPLAY_FIXTURE}"
                    --out "${CMAKE_CURRENT_BINARY_DIR}/gallery-ui")
                set_tests_properties(gallery_ui PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 360 RUN_SERIAL TRUE)
            endif()
        endif()
        if(MAPX_BUILD_DEVELOPER_TOOLS AND MAPX_FIXTURE_MANIFEST_DIR AND MAPX_FIXTURE_BASE)
            add_test(NAME input_fixtures COMMAND map-extractor-probe --check
                "${MAPX_FIXTURE_MANIFEST_DIR}/fixtures.txt" "${MAPX_FIXTURE_BASE}")
            foreach(kind IN ITEMS pre_session extract generate)
                if(kind STREQUAL "pre_session")
                    set(mode --pre-check)
                elseif(kind STREQUAL "extract")
                    set(mode --extract-check)
                else()
                    set(mode --gen-check)
                endif()
                add_test(NAME ${kind}_fixtures COMMAND map-extractor-probe ${mode}
                    "${MAPX_FIXTURE_MANIFEST_DIR}/${kind}_fixtures.txt" "${MAPX_FIXTURE_BASE}" "${MAPX_GAME_ROOT}")
            endforeach()
        endif()
    endif()
endif()
